```mermaid
classDiagram
  namespace target {
    class IShape {
      <<interface>>
      GetStyle() IStyle&
    }

    class IStyle {
      <<interface>>
      GetColor() uint32
    }
  }

  namespace adaptee {
    class Shape {
      m_style: Style
      GetStyle() Style&
    }
    class Style {
      m_color: Color
      GetColor() Color
    }
    class Color {
      <<struct>>
      r: byte
      g: byte
      b: byte
      a: byte
    }
  }
  class Client {
  }

  IShape ..> IStyle
  Shape *-- Style
  Style *-- Color
  Client ..> IShape
  Client ..> IStyle
```
